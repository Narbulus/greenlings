# One demonstration teaches every greenling that is watching.
!include _setup.txt
gldev_place piece_chest_wood -4 4
gldev_place piece_chest 5 8
gldev_fill piece_chest_wood Wood 10
gldev_give Coins 5
gldev_place Greyling -1 3
!wait 1
gldev_tame
gldev_bond 100
gldev_useitem Coins
gldev_pat
gldev_place Greyling 1 3
!wait 1
gldev_tame
gldev_bond 100
gldev_useitem Coins
gldev_pat
!waitfor GLEVT\] state .*to=Watching 3
!waitfor GLEVT\] state .*to=Watching 3
gldev_open piece_chest_wood
!wait 0.5
gldev_take Wood 1
gldev_close
gldev_open piece_chest
!wait 0.5
gldev_put Wood 1
gldev_close
!waitfor GLEVT\] learned gl=(\w+)#\d+ .*Wood: 3
!waitfor GLEVT\] learned gl=(\w+)#\d+ .*Wood: 3
!waitfor GLEVT\] deposit .*item=Wood 40
