# Upkeep: greenlings help themselves to food from a chest when hungry, and sleep in a nest at night.
!include _setup.txt
tod 0.5
!include _tamed.txt
!wait 2.5
gl_dump 10
!waitfor GLDUMP\] \{"type":"greenling".*"needs":"no nest" 2
# hungry now: it finds the food chest
gldev_place piece_chest_wood 3 4
gldev_fill piece_chest_wood Raspberry 5
gldev_cfg greenlings Needs MealInterval 1
!waitfor GLEVT\] meal_seek .*item=Raspberry 10
!waitfor GLEVT\] meal .*item=Raspberry 30
gldev_cfg greenlings Needs MealInterval 900
# a nest makes it a home; at night it goes to bed, in the morning it gets up
gldev_place piece_greenling_nest -3 5
!wait 3
gl_dump 10
!waitfor GLDUMP\] \{"type":"greenling".*"needs":"" 2
tod 0
!waitfor GLEVT\] sleep nest= 10
!wait 3
tod 0.5
!waitfor GLEVT\] woke 10
tod -1
