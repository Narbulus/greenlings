# Take finished food off a cooking station and store it, before it burns.
!include _setup.txt
gldev_place piece_chest_wood -3 4
gldev_place fire_pit 3 6
gldev_place piece_cookingstation 3 6
gldev_give Wood 3
gldev_use fire_pit Wood
gldev_give RawMeat 2
gldev_use piece_cookingstation RawMeat
gldev_use piece_cookingstation RawMeat
gldev_move 0 -5
!include _watching.txt
# cook (vanilla raw meat takes ~25s); keep the greenling's attention while we wait
gldev_cfg greenlings Teaching AttentionSpan 120
!wait 30
gl_dump 15
gldev_interact piece_cookingstation
!wait 1
gldev_pickup CookedMeat
!waitfor GLEVT\] demo_source item=CookedMeat at=Extract:piece_cookingstation 3
gldev_open piece_chest_wood
!wait 0.5
gldev_put CookedMeat 1
gldev_close
!waitfor GLEVT\] learned .*CookedMeat: Extract:piece_cookingstation#\d+ -> Container:piece_chest_wood 3
!waitfor GLEVT\] harvest .*item=CookedMeat 30
!waitfor GLEVT\] take .*item=CookedMeat 20
!waitfor GLEVT\] deposit .*item=CookedMeat 30
gldev_cfg greenlings Teaching AttentionSpan 60
gl_dump 15
