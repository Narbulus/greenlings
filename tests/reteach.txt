# A working greenling can be taught a new job, and told to forget it.
!include _setup.txt
gldev_place piece_chest_wood -4 4
gldev_place piece_chest 5 8
gldev_place smelter -6 10
gldev_fill piece_chest_wood Wood 50
gldev_fill piece_chest_wood CopperOre 20
!include _watching_dwarf.txt
# job 1: wood chest -> chest
gldev_open piece_chest_wood
!wait 0.5
gldev_take Wood 1
gldev_close
gldev_open piece_chest
!wait 0.5
gldev_put Wood 1
gldev_close
!waitfor GLEVT\] learned .*Wood: 3
!waitfor GLEVT\] take .*item=Wood 30
# re-teach while it works: copper ore -> smelter
gldev_useitem Coins
!waitfor GLEVT\] state .*from=Working to=Watching 3
gldev_open piece_chest_wood
!wait 0.5
gldev_take CopperOre 1
gldev_close
gldev_addore smelter CopperOre
!waitfor GLEVT\] learned .*CopperOre: Container:piece_chest_wood#\d+ -> SmelterOre:smelter 3
!waitfor GLEVT\] take .*item=CopperOre 30
# and forget it again: a coin to get its attention, a pat to say never mind
gldev_useitem Coins
!waitfor GLEVT\] state .*to=Watching 3
gldev_pat
!waitfor GLEVT\] forgot .*CopperOre 3
!waitfor GLEVT\] state .*to=Idle reason=dismissed 3
gl_dump
